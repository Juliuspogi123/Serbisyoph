${resumeData.skills}

              <Plus className="h-4 w-4 mr-1"/> Add Education
            </button>

            <h3 className="text-xl font-semibold mb-4 text-blue-700 border-b pb-2">Skills</h3>
            <textarea value={resumeData.skills} onChange={(e) => handleUpdate('skills', e.target.value)} rows="3" className="w-full border border-gray-300 rounded-md p-2" placeholder="List your skills separated by commas (e.g. Communication, Microsoft Office, Problem Solving)..."></textarea>
          </div>

          <div className="flex flex-col">
            <div className="flex justify-end space-x-2 mb-4 print:hidden">
              <button onClick={handleCopy} className="px-4 py-2 bg-gray-800 text-white rounded-md flex items-center hover:bg-gray-700 shadow-sm transition-colors">
                <Copy className="h-4 w-4 mr-2" /> Copy Text

                <h2 className="text-xl text-gray-600 mb-3">{resumeData.title || 'Professional Title'}</h2>
                <div className="flex flex-wrap justify-center gap-4 text-sm text-gray-600">
                  {resumeData.email && <span className="flex items-center"><Mail className="h-3 w-3 mr-1" /> {resumeData.email}</span>}
                  {resumeData.phone && <span className="flex items-center"><Phone className="h-3 w-3 mr-1" /> {resumeData.phone}</span>}
                  {resumeData.location && <span className="flex items-center"><MapPin className="h-3 w-3 mr-1" /> {resumeData.location}</span>}
                </div>
              </div>

              {/* Summary */}
              {resumeData.summary && (
                <div className="mb-6">
                  <h3 className="text-lg font-bold uppercase border-b border-gray-300 mb-2">Professional Summary</h3>
                  <p className="text-sm leading-relaxed">{resumeData.summary}</p>
                </div>
              )}

              {/* Experience */}
              {resumeData.experience.some(e => e.title || e.company) && (
                <div className="mb-6">
                  <h3 className="text-lg font-bold uppercase border-b border-gray-300 mb-2">Experience</h3>
                  <div className="space-y-4">
                    {resumeData.experience.filter(e => e.title || e.company).map((exp, idx) => (
                      <div key={idx}>
                        <div className="flex justify-between font-bold">
                          <span className="text-[15px]">{exp.title}</span>
                          <span className="text-sm">{exp.year}</span>
                        </div>
                        <div className="text-sm italic text-gray-700 mb-1">{exp.company}</div>
                        <p className="text-sm">{exp.desc}</p>
                      </div>
                    ))}
                  </div>
                </div>
              )}

              {/* Education */}
              {resumeData.education.some(e => e.degree || e.school) && (
                <div className="mb-6">
                  <h3 className="text-lg font-bold uppercase border-b border-gray-300 mb-2">Education</h3>
                  <div className="space-y-3">
                    {resumeData.education.filter(e => e.degree || e.school).map((edu, idx) => (
                      <div key={idx}>
                        <div className="flex justify-between font-bold">
                          <span className="text-[15px]">{edu.degree}</span>
                          <span className="text-sm">{edu.year}</span>
                        </div>
                        <div className="text-sm text-gray-700">{edu.school}</div>
                      </div>
                    ))}
                  </div>
                </div>
              )}

              {/* Skills */}
              {resumeData.skills && (
                <div>
                  <h3 className="text-lg font-bold uppercase border-b border-gray-300 mb-2">Skills</h3>
                  <p className="text-sm leading-relaxed">{resumeData.skills}</p>
                </div>
              )}
            </div>
          </div>
        </div>
      </div>
    );
  };

  const GovGuidesView = () => {
    const [selectedGuide, setSelectedGuide] = useState(null);

    if (selectedGuide) {
      return (
        <div className="max-w-4xl mx-auto px-4 py-8 animate-in slide-in-from-right-4 duration-300">
          <button onClick={() => setSelectedGuide(null)} className="mb-6 text-blue-600 font-medium hover:underline flex items-center">
            ← Back to Directory
          </button>
          
          <div className="bg-white rounded-xl shadow-lg p-6 md:p-10 border-t-4 border-yellow-500 relative overflow-hidden">
            <ShieldCheck className="absolute -top-4 -right-4 h-32 w-32 text-gray-50 opacity-50" />
            
            <div className="flex items-center mb-2">
              <span className="bg-blue-100 text-blue-800 text-xs font-semibold px-2.5 py-0.5 rounded uppercase tracking-wide">
                {selectedGuide.category}
              </span>
            </div>
            
            <h2 className="text-3xl font-bold text-gray-900 mb-4">{selectedGuide.title} Guide</h2>
            <p className="text-lg text-gray-700 mb-8 border-l-4 border-blue-500 pl-4">{selectedGuide.purpose}</p>

            <div className="grid md:grid-cols-2 gap-8">
              <div>
                <h3 className="text-xl font-bold text-gray-800 mb-4 flex items-center border-b pb-2">
                  <FileText className="h-5 w-5 mr-2 text-blue-600" /> Primary Requirements
                </h3>
                <ul className="space-y-2">
                  {selectedGuide.requirements.map((req, idx) => (
                    <li key={idx} className="flex items-start">
                      <span className="text-yellow-500 mr-2 font-bold">•</span>
                      <span className="text-gray-700">{req}</span>
                    </li>
                  ))}
                </ul>
              </div>

              <div>
                <h3 className="text-xl font-bold text-gray-800 mb-4 flex items-center border-b pb-2">
                  <ExternalLink className="h-5 w-5 mr-2 text-blue-600" /> Official Registration
                </h3>
                <p className="text-sm text-gray-600 mb-4">Always transact directly through the official government portal to avoid scams.</p>
                <a href={selectedGuide.link} target="_blank" rel="noopener noreferrer" className="inline-flex items-center px-4 py-3 bg-blue-600 hover:bg-blue-700 text-white rounded-lg font-medium shadow-md transition-colors w-full justify-center">
                  Visit Official Website <ExternalLink className="h-4 w-4 ml-2" />
                </a>
              </div>
            </div>

            <div className="mt-8 bg-gray-50 p-6 rounded-lg border border-gray-200">
              <h3 className="text-xl font-bold text-gray-800 mb-4">General Steps</h3>
              <ol className="list-decimal list-inside space-y-3">
                {selectedGuide.steps.map((step, idx) => (
                  <li key={idx} className="text-gray-700 pl-2 leading-relaxed">{step}</li>
                ))}
              </ol>
            </div>
          </div>
        </div>
      );
    }

    return (
      <div className="max-w-7xl mx-auto px-4 py-8 animate-in fade-in">
        <div className="text-center mb-10">
          <h2 className="text-3xl font-bold text-gray-900 mb-4">Government Online Guides</h2>
          <p className="text-gray-600 max-w-2xl mx-auto">Find simplified instructions on how to register, apply, and transact with major Philippine government agencies online.</p>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
          {GOV_GUIDES.map(guide => (
            <div key={guide.id} onClick={() => setSelectedGuide(guide)} className="bg-white rounded-lg shadow-sm border border-gray-200 p-6 hover:shadow-md hover:border-blue-300 transition-all cursor-pointer flex flex-col h-full">
               <div className="text-xs font-semibold text-blue-600 mb-2 uppercase tracking-wider">{guide.category}</div>
               <h3 className="text-xl font-bold text-gray-800 mb-2">{guide.title}</h3>
               <p className="text-gray-600 text-sm mb-4 flex-grow line-clamp-2">{guide.purpose}</p>
               <div className="text-yellow-600 text-sm font-medium flex items-center mt-auto">
                 Read Guide <ChevronRight className="h-4 w-4 ml-1" />
               </div>
            </div>
          ))}
        </div>
      </div>
    );
  };

  const JobAssistantView = () => {
    const [openSection, setOpenSection] = useState('templates');

    const toggleSection = (section) => {
      setOpenSection(openSection === section ? null : section);
    };

    const AccordionHeader = ({ id, title, icon: Icon }) => (
      <button 
        onClick={() => toggleSection(id)}
        className={`w-full flex items-center justify-between p-4 bg-white hover:bg-gray-50 border-b border-gray-200 transition-colors ${openSection === id ? 'bg-blue-50' : ''}`}
      >
        <div className="flex items-center text-lg font-semibold text-gray-800">
          <Icon className={`h-6 w-6 mr-3 ${openSection === id ? 'text-blue-600' : 'text-gray-500'}`} />
          {title}
        </div>
        {openSection === id ? <ChevronUp className="h-5 w-5 text-gray-500" /> : <ChevronDown className="h-5 w-5 text-gray-500" />}
      </button>
    );

    return (
      <div className="max-w-4xl mx-auto px-4 py-8 animate-in fade-in">
         <div className="text-center mb-10">
          <h2 className="text-3xl font-bold text-gray-900 mb-4">Job Seeking Assistant</h2>
          <p className="text-gray-600 max-w-2xl mx-auto">Tools and resources to help you land your next job, from crafting emails to acing the interview.</p>
        </div>

        <div className="bg-white rounded-xl shadow-md border border-gray-200 overflow-hidden">
          
          {/* Section 1: Platforms */}
          <AccordionHeader id="platforms" title="Where to find jobs online" icon={Briefcase} />
          {openSection === 'platforms' && (
            <div className="p-6 bg-gray-50 text-gray-700">
              <p className="mb-4">Here are some of the most reliable platforms for finding jobs in the Philippines:</p>
              <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
                <a href="https://www.jobstreet.com.ph/" target="_blank" rel="noopener noreferrer" className="flex items-center justify-center p-4 bg-white border border-gray-300 rounded-lg hover:border-blue-500 hover:text-blue-600 transition">JobStreet</a>
                <a href="https://www.linkedin.com/" target="_blank" rel="noopener noreferrer" className="flex items-center justify-center p-4 bg-white border border-gray-300 rounded-lg hover:border-blue-500 hover:text-blue-600 transition">LinkedIn</a>
                <a href="https://ph.indeed.com/" target="_blank" rel="noopener noreferrer" className="flex items-center justify-center p-4 bg-white border border-gray-300 rounded-lg hover:border-blue-500 hover:text-blue-600 transition">Indeed</a>
              </div>
              <p className="mt-4 text-sm italic text-gray-500">Note: Always verify the employer profile before applying.</p>
            </div>
          )}

          {/* Section 2: Templates */}
          <AccordionHeader id="templates" title="Application Email Templates" icon={FileText} />
          {openSection === 'templates' && (
            <div className="p-6 bg-gray-50 space-y-6">
              <p className="text-gray-700">Copy these templates and replace the bracketed text [like this] with your own information.</p>
              {TEMPLATES.map((tmpl, idx) => (
                <div key={idx} className="bg-white p-4 rounded-lg border border-gray-200 shadow-sm relative group">
                  <h4 className="font-bold text-gray-800 mb-2">{tmpl.title}</h4>
                  <pre className="whitespace-pre-wrap font-sans text-sm text-gray-600 bg-gray-100 p-4 rounded-md">
                    {tmpl.content}
                  </pre>
                  <button 
                    onClick={() => { navigator.clipboard.writeText(tmpl.content); alert('Template copied!'); }}
                    className="absolute top-4 right-4 p-2 bg-blue-100 text-blue-700 rounded hover:bg-blue-200 opacity-0 group-hover:opacity-100 transition-opacity flex items-center text-xs font-semibold"
                  >
                    <Copy className="h-3 w-3 mr-1" /> Copy
                  </button>
                </div>
              ))}
            </div>
          )}

          {/* Section 3: Interview Prep */}
          <AccordionHeader id="interview" title="Interview Preparation" icon={Phone} />
          {openSection === 'interview' && (
            <div className="p-6 bg-gray-50 space-y-6">
              <p className="text-gray-700 mb-4">Practice these common questions to build your confidence.</p>
              {INTERVIEW_PREP.map((item, idx) => (
                <div key={idx} className="bg-white p-4 rounded-lg border border-l-4 border-l-yellow-500 border-y-gray-200 border-r-gray-200 shadow-sm">
                  <span className="text-xs font-bold text-gray-400 uppercase tracking-wider">{item.category}</span>
                  <h4 className="font-bold text-gray-900 mt-1 mb-2">Q: {item.q}</h4>
                  <p className="text-gray-700 text-sm leading-relaxed"><strong className="text-blue-700 font-semibold">Tip:</strong> {item.a}</p>
                </div>
              ))}
            </div>
          )}

          {/* Section 4: Scam Awareness */}
          <AccordionHeader id="scams" title="Job Scam Awareness" icon={AlertTriangle} />
          {openSection === 'scams' && (
            <div className="p-6 bg-red-50 text-red-900 border-t border-red-100">
              <h4 className="font-bold text-xl mb-4 flex items-center"><AlertTriangle className="h-6 w-6 mr-2 text-red-600"/> Red Flags of Fake Jobs</h4>
              <ul className="space-y-3 list-disc list-inside">
                <li><strong>Requests for payment:</strong> Legitimate employers will never ask you to pay for training, equipment, or application fees upfront.</li>
                <li><strong>Unrealistic salary:</strong> "Earn 50k a week typing from home" is almost always a scam.</li>
                <li><strong>Asking for OTPs or Passwords:</strong> Never share your bank passwords or One-Time Pins (OTPs) with a "recruiter".</li>
                <li><strong>Suspicious Links:</strong> Be wary of links sent via SMS or Telegram that look unofficial (e.g., sss-gov-ph-update.com instead of sss.gov.ph).</li>
              </ul>
              <p className="mt-6 font-bold text-red-700">When in doubt, search for the company online and contact them directly through their official website to verify the job opening.</p>
            </div>
          )}

        </div>
      </div>
    );
  };

  const SearchResultsView = () => {
    const query = searchQuery.toLowerCase();
    
    const matchedGuides = GOV_GUIDES.filter(g => 
      g.title.toLowerCase().includes(query) || 
      g.purpose.toLowerCase().includes(query) ||
      g.category.toLowerCase().includes(query)
    );

    const matchedInterview = INTERVIEW_PREP.filter(i =>
      i.q.toLowerCase().includes(query) ||
      i.category.toLowerCase().includes(query)
    );

    return (
      <div className="max-w-7xl mx-auto px-4 py-8 animate-in fade-in">
         <div className="mb-8">
            <button onClick={() => { setSearchQuery(''); setCurrentTab('home'); }} className="text-blue-600 font-medium hover:underline mb-4 inline-block">
              ← Back to Home
            </button>
            <h2 className="text-2xl font-bold text-gray-900">Search Results for "{searchQuery}"</h2>
         </div>

         {matchedGuides.length === 0 && matchedInterview.length === 0 && (
           <div className="text-center py-12 bg-white rounded-lg shadow-sm border border-gray-200">
             <Search className="h-12 w-12 text-gray-300 mx-auto mb-4" />
             <h3 className="text-lg font-medium text-gray-900">No exact matches found</h3>
             <p className="text-gray-500 mt-2">Try searching for keywords like "SSS", "Resume", "Interview", or "NBI".</p>
           </div>
         )}

         {matchedGuides.length > 0 && (
           <div className="mb-8">
             <h3 className="text-xl font-bold text-gray-800 mb-4 border-b pb-2">Government Guides</h3>
             <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                {matchedGuides.map(guide => (
                  <div key={guide.id} onClick={() => { setSearchQuery(''); setCurrentTab('gov'); }} className="bg-white p-4 rounded-lg shadow-sm border border-gray-200 hover:border-blue-400 cursor-pointer">
                    <h4 className="font-bold text-blue-700">{guide.title}</h4>
                    <p className="text-sm text-gray-600 mt-1 line-clamp-2">{guide.purpose}</p>
                  </div>
                ))}
             </div>
           </div>
         )}

         {matchedInterview.length > 0 && (
           <div>
             <h3 className="text-xl font-bold text-gray-800 mb-4 border-b pb-2">Interview Preparation</h3>
             <div className="space-y-4">
                {matchedInterview.map((item, idx) => (
                  <div key={idx} className="bg-white p-4 rounded-lg shadow-sm border border-gray-200">
                    <h4 className="font-bold text-gray-900">Q: {item.q}</h4>
                    <p className="text-sm text-gray-700 mt-2">{item.a}</p>
                  </div>
                ))}
             </div>
           </div>
         )}
      </div>
    );
  };

  // --- RENDER MAIN LAYOUT ---
  return (
    <div className="min-h-screen bg-gray-50 font-sans text-gray-900 selection:bg-yellow-200 selection:text-gray-900">
      <Header />
      
      <main className="print:m-0 print:p-0">
        {currentTab === 'search_results' && searchQuery.trim() !== '' ? (
          <SearchResultsView />
        ) : (
          <>
            {currentTab === 'home' && <HomeView />}
            {currentTab === 'resume' && <ResumeBuilderView />}
            {currentTab === 'gov' && <GovGuidesView />}
            {currentTab === 'jobs' && <JobAssistantView />}
          </>
        )}
      </main>

      <Footer />
    </div>
  );
}
